## Module Examples

### Basic Usage

```javascript
// Memory
const Memory = require('./30_long-term-memory');

// Reasoning
const Reasoning = require('./42_reasoning-module');

reasoner.analyze('All humans are mortal. Socrates is human.');

const Emotion = require('./18_emotional-processing');
```
